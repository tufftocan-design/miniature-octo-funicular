local UserInputService = game:GetService("UserInputService")

local player = Players.LocalPlayer
local isZombieMode = false
local settingsOpen = false

local CONFIG_PATH = "ZombieAnimConfig.json"

local ANIM_STATES = {"Idle","Idle2","Walk","Run","Jump","Fall","Climb","Swim","SwimIdle"}

local ZOMBIE_ANIMS = {
	Idle = "rbxassetid://616158929",
	Idle2 = "rbxassetid://616160636",
	Walk = "rbxassetid://616168032",
	Run = "rbxassetid://616163682",
	Jump = "rbxassetid://616161997",
	Fall = "rbxassetid://616157476",
	Climb = "rbxassetid://616165109",
	Swim = "rbxassetid://616165109",
	SwimIdle = "rbxassetid://616166655",
}

local FOLDER_MAP = {
	Idle = "idle", Idle2 = "idle", Walk = "walk", Run = "run",
	Jump = "jump", Fall = "fall", Climb = "climb", Swim = "swim", SwimIdle = "swimidle",
}

local DEFAULT_ANIMS = {}
local enabledStates = {}

local function loadConfig()
	if isfile and isfile(CONFIG_PATH) then
		local ok, data = pcall(function()
			return HttpService:JSONDecode(readfile(CONFIG_PATH))
		end)
		if ok and type(data) == "table" then
			for _, state in ipairs(ANIM_STATES) do
				enabledStates[state] = data[state]
				if enabledStates[state] == nil then enabledStates[state] = true end
			end
			return
		end
	end
	for _, state in ipairs(ANIM_STATES) do
		enabledStates[state] = true
	end
end

local function saveConfig()
	if writefile then
		writefile(CONFIG_PATH, HttpService:JSONEncode(enabledStates))
	end
end

loadConfig()

local function syncDropdownPosition()
	local absPos = mainFrame.AbsolutePosition
	local absSize = mainFrame.AbsoluteSize
	dropFrame.Position = UDim2.new(0, absPos.X, 0, absPos.Y + absSize.Y + 6)
end

mainFrame:GetPropertyChangedSignal("Position"):Connect(syncDropdownPosition)
mainFrame:GetPropertyChangedSignal("AbsolutePosition"):Connect(syncDropdownPosition)

local dragging = false
local dragStart, startPos

mainFrame.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		dragging = true
		dragStart = input.Position
		startPos = mainFrame.Position
	end
end)

UserInputService.InputChanged:Connect(function(input)
	if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
		local delta = input.Position - dragStart
		mainFrame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
		syncDropdownPosition()
	end
end)

mainFrame.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		dragging = false
	end
end)

local function getAnimIdInFolder(folder, folderName)
	local sub = folder:FindFirstChild(folderName)
	if sub then
		for _, anim in ipairs(sub:GetChildren()) do
			if anim:IsA("Animation") then
				return anim.AnimationId
			end
		end
	end
	return nil
end

local function setAnim(folder, folderName, animId)
	local sub = folder:FindFirstChild(folderName)
	if sub then
		local applied = false
		for _, anim in ipairs(sub:GetChildren()) do
			if anim:IsA("Animation") then
				anim.AnimationId = animId
				applied = true
			end
		end
		if not applied then
			local newAnim = Instance.new("Animation")
			newAnim.AnimationId = animId
			newAnim.Parent = sub
		end
		return true
	end
	return false
end

local function captureDefaults(animate)
	for _, state in ipairs(ANIM_STATES) do
		if DEFAULT_ANIMS[state] == nil then
			local folderName = FOLDER_MAP[state]
			local id = getAnimIdInFolder(animate, folderName)
			if id then
				DEFAULT_ANIMS[state] = id
			end
		end
	end
end

local function applyZombieAnims(animate)
	if not animate then return false end
	captureDefaults(animate)
	local ok = false
	for _, state in ipairs(ANIM_STATES) do
		local folderName = FOLDER_MAP[state]
		local targetId
		if enabledStates[state] then
			targetId = ZOMBIE_ANIMS[state]
		else
			targetId = DEFAULT_ANIMS[state]
		end
		if targetId then
			if setAnim(animate, folderName, targetId) then ok = true end
		end
	end
	return ok
end

local function activateZombieMode()
	if isZombieMode then return end

	local char = player.Character
	if not char then return end
	local animate = char:FindFirstChild("Animate")
	if not animate then return end

	local success = applyZombieAnims(animate)

	if success then
		isZombieMode = true
		toggleButton.BackgroundColor3 = Color3.fromRGB(0, 170, 70)
	end
end

local function refreshCurrentAnims()
	if not isZombieMode then return end
	local char = player.Character
	if not char then return end
	local animate = char:FindFirstChild("Animate")
	if not animate then return end
	applyZombieAnims(animate)
end

		refreshVisual()
		saveConfig()
		refreshCurrentAnims()
	end)
end
